				local attachment2 = {}

				end

		end

function module3.setupHideVisualsOnMoveDir(p52)

		forward = {},

		local v248 = getDirectionVec3()
		local s1 = nil

		if v248.Z > 0.5 then
			s1 = "backward"

			s1 = "forward"
		elseif v248.X > 0.5 then

		end

				for _, v in pairs(t10[s1]) do
					v:SetAttribute("EnabledCount", v:GetAttribute("EnabledCount") - 1)
					table.insert(t11, v)
				end

					for _, v in pairs(t11) do
						v:SetAttribute("EnabledCount", v:GetAttribute("EnabledCount") + 1)
					end
				end
			end
		else
			module.reverseDisableOnMoveDirection = nil
		end

		u138 = s1
	end

	local v140 = false

	for _, visual12 in inst:QueryDescendants("Trail") do
		local WorldCFrame = visual12.Attachment0.WorldCFrame
		local Position = inst.CFrame:ToObjectSpace(WorldCFrame).Position

		if Position.Z > 0 then
			s2 = "backward"
		elseif Position.Z < 0 then
			s2 = "forward"
		elseif Position.X > 0 then
			s2 = "right"
		elseif Position.X < 0 then
			s2 = "left"
		end

		if s2 then
			v140 = true
			table.insert(t10[s2], visual12)
		end
	end

	if v140 then
		module:GiveTask(hookMoveDirectionChanged(onMoveDirectionChanged))
	end

function module3.rigVFX(part2, template3, p65)

		v173 *= p65.relativeRootPart.CFrame:ToObjectSpace(clone:GetPivot())
	end

	if p65.createAttachment then
		Attachment = Instance.new("Attachment")
		Attachment.Parent = parent or part2
