local module = {}
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local TweenService = game:GetService("TweenService")
local VisualEffectsUtil = require(ReplicatedStorage.Shared.Source.VFX.VisualEffectsUtil)

local function tweenSingleBeamTransparency(p1, p2, p3, p4, p5, p6)
	local Keypoints = p1.Transparency.Keypoints
	local v14 = #Keypoints
	local v15 = nil
	local v16 = nil

	if typeof(p2) == "number" then
		v15 = p2
		v16 = p2
	elseif typeof(p2) == "string" then
		local v17, v18 = p2:match("([^,]+),([^,]+)")

		v15 = tonumber(v17)
		v16 = tonumber(v18)
	end

	if not v15 then
		return nil
	end

	if p1.Enabled then
		return nil
	end

	p1.Enabled = true

	local TextureSpeed = p1.TextureSpeed
	local p1Transparency = p1.Transparency
	local create3 = table.create(v14)
	local create2 = table.create(v14)

	for v23, v24 in Keypoints do
		create3[v23] = v24.Value
		create2[v23] = v15 + (v16 - v15) * v24.Time
	end

	TweenService:Create(p1, TweenInfo.new(p3, p4, p5), {
		TextureSpeed = p6
	}):Play()

	local n1 = 0
	local connection = nil

	connection = RunService.Heartbeat:Connect(function(dt)
		n1 += dt

		if n1 > p3 then
			connection:Disconnect()

			return
		end

		local Value = TweenService:GetValue(math.clamp(n1 / p3, 0, 1), p4, p5)
		local create = table.create(v14)

		for new, v53 in Keypoints do
			local v54 = create3[new] + (create2[new] - create3[new]) * Value

			create[new] = NumberSequenceKeypoint.new(v53.Time, v54, v53.Envelope)
		end

		p1.Transparency = NumberSequence.new(create)
	end)

	return function()
		if connection then
			connection:Disconnect()
		end

		p1.Enabled = false
		p1.TextureSpeed = TextureSpeed
		p1.Transparency = p1Transparency
	end
end

function module.toggleAttachment(p7, p8, p9)
	for _, v31 in p7:QueryDescendants("ParticleEmitter") do
		v31.Enabled = p8
	end

	if p9 and p9 > 0 then
		task.delay(p9, module.toggleAttachment, p7, false)
	end
end
function module.emitAttachment(p10)
	for _, instance in p10:QueryDescendants("ParticleEmitter") do
		local v35 = instance:GetAttribute("EmitCount") or 1
		local v36 = instance:GetAttribute("EmitDelay") or 0
		local amount = math.max(v35 * VisualEffectsUtil.getEmitCountOptimizeMult(instance.Parent), 1)

		if v36 > 0 then
			task.delay(v36, instance.Emit, instance, amount)
		else
			instance:Emit(amount)
		end
	end
end
function module.setTimeScale(p11, p12)
	for _, v41 in p11:QueryDescendants("ParticleEmitter") do
		v41.TimeScale = p12
	end
end
function module.tweenBeamTransparency(p13, p14)
	local v44 = p14 or {}
	local duration = v44.duration or 2

	for _, v47 in p13:QueryDescendants("Beam") do
		local v48 = tweenSingleBeamTransparency(v47, v44.transparency or 1, duration, v44.beamEasingStyle or Enum.EasingStyle.Sine, v44.beamEasingDirection or Enum.EasingDirection.Out, v44.speed or 0)

		if v48 then
			task.delay(duration + 0.15, v48)
		end
	end
end

return module
