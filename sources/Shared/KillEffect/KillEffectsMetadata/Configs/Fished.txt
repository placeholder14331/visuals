local KillEffectUtil = require(game.ReplicatedStorage.Shared.Source.KillEffect.KillEffectUtil)
local Fished = game.ReplicatedStorage.Shared.Assets.Cosmetics.KillEffects.Fished
local createPartCacheFromTemplate = KillEffectUtil.createPartCacheFromTemplate(Fished.Mackerel, 20)

return function()
	return {
		displayName = "Fish'd",
		icon = "rbxassetid://107040245180838",
		vaulted = false,
		refundIfVaulted = false,
		rarity = "epic",
		inItemShop = true,
		apply = function(character)
			local KillEffectsUtilClient = require(game.ReplicatedStorage.Client.Source.KillEffect.KillEffectsUtilClient)
			local HumanoidRootPart = character.HumanoidRootPart
			local Part = createPartCacheFromTemplate:GetPart()

			task.delay(4.5, function()
				local part = Part

				part.Size = part.Size / 4
				Part.CanCollide = true
				createPartCacheFromTemplate:ReturnPart(Part)
			end)
			Part.CFrame = CFrame.new(HumanoidRootPart.Position + Vector3.new(0, 2, 0)) * CFrame.Angles(0, 0, 1.57079633)
			Part.Size = Part.Size * 4
			Part.Anchored = true
			Part.CanCollide = false
			KillEffectsUtilClient.replaceCharacterWithObjects(character, createPartCacheFromTemplate, {
				amount = 4
			})
		end
	}
end
