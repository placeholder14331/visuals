local ReplicatedStorage = game:GetService("ReplicatedStorage")
local VisualEffectsUtil = require(ReplicatedStorage.Shared.Source.VFX.VisualEffectsUtil)
local KillEffectsUtilClient = nil
local AuroraBurst = game.ReplicatedStorage.Shared.Assets.Cosmetics.KillEffects.AuroraBurst

return function()
	return {
		displayName = "Northern Aurora Burst",
		icon = "rbxassetid://87896133517919",
		vaulted = false,
		refundIfVaulted = false,
		rarity = "rare",
		caseIds = { "case16" },
		willCharacterDisappear = true,
		apply = function(character)
			KillEffectsUtilClient = require(ReplicatedStorage.Client.Source.KillEffect.KillEffectsUtilClient)

			local HumanoidRootPart = character.HumanoidRootPart
			local part = KillEffectsUtilClient.getCharacterFloor(character)
			local position = part and part.Position or HumanoidRootPart.Position + Vector3.new(0, -2.5, 0)
			local clone = AuroraBurst.Ground:Clone()

			task.delay(4, function()
				clone:Destroy()
			end)
			clone.Parent = HumanoidRootPart
			clone.WorldPosition = position

			return VisualEffectsUtil.emitObject(clone)
		end
	}
end
