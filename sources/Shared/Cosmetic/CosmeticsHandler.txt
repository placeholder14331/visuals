function module.getIsCustomCosmeticWordOk(p9)
	local lower = string.lower(p9)

	if t2[lower] then
		return false
	end

	for k in pairs(t2) do
		if string.find(lower, " " .. k) then
			return false
		end

		if string.find(lower, k .. " ") then
			return false
		end
	end

	return true
end

		local metadata = metadata3[metadata2].metadata

		if metadata then
			local shallowCopyTable = TableUtil.shallowCopyTable(metadata)

			return shallowCopyTable
		end
	end
end

function module.getCosmeticUnlockPrerequeistesFulfilled(p14, p15, p16, p17)
	local getDataRoduxStoreForPlayer2 = DataHandler.getDataRoduxStoreForPlayer(p14):getState()
	local getSessionStore = StateHandler.getSessionStore(p14)
	local getCosmeticConfig = module.getCosmeticConfig(p15, p16)
	local v54 = getSessionStore.quests.currentCosmeticQuestIds()[p15]

	if not p17 and (getCosmeticConfig and getCosmeticConfig.isOffsale) then
		return false, getCosmeticConfig.offsaleReason or "This item is offsale"
	end

	if not p17 and getCosmeticConfig and getCosmeticConfig.disabled then
		return false, getCosmeticConfig.disabledReason or "This item is disabled"
	end

	if getCosmeticConfig and getCosmeticConfig.vaulted then
		return false, "This item is vaulted"
	end

	if getCosmeticConfig and getCosmeticConfig.isAvailableInPublicTesting then
		return true
	end

	if v54 then
		return false, "This item has an active quest"
	end

	if getCosmeticConfig and getCosmeticConfig.unlockFirst then
		for _, v in pairs(getCosmeticConfig.unlockFirst) do
			local itemId = v.itemId
			local itemType = v.itemType
			local typeConfigs2 = ItemGroups.typeConfigs[itemType]
			local configsModulePath3 = require(typeConfigs2.configsModulePath)[itemId]

			if not getDataRoduxStoreForPlayer2[typeConfigs2.ownedThingsId][itemId] then
				return false, `{ configsModulePath3.displayName } must be unlocked first`
			end
		end
	end

	return true
end

	end

end
