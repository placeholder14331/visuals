local CosmeticGroups = require(game.ReplicatedStorage.Shared.Source.Cosmetic.CosmeticGroups)

return {
	getCosmeticIdsWithRarity = function(rarity, p2)
		local module = {}

		for k, v in pairs(require(CosmeticGroups.typeConfigs[p2].configsModulePath)) do
			if rarity == v.rarity then
				table.insert(module, k)
			end
		end

		return module
	end
}
