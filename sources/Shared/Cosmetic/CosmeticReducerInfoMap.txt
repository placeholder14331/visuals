local Rodux = require(game.ReplicatedStorage.Shared.Vendor.Rodux)
local CosmeticGroups = require(game.ReplicatedStorage.Shared.Source.Cosmetic.CosmeticGroups)
local typeConfigs = CosmeticGroups.typeConfigs
local individualTypeConfigs = CosmeticGroups.individualTypeConfigs
local DefaultData = require(game.ReplicatedStorage.Shared.Source.Data.DefaultData)
local TableUtil = require(game.ReplicatedStorage.Shared.Vendor.Table.TableUtil)
local ThingUtil = require(game.ReplicatedStorage.Shared.Source.Thing.ThingUtil)

local function getNewOwnedCosmeticIdAmount(p1)
	if p1 <= 0 then
		return nil
	end

	return p1
end
local function createOwnedCosmeticsReducerInfo(p2)
	local ownedThingsId = typeConfigs[p2].ownedThingsId
	local module = {}
	local v25 = DefaultData[ownedThingsId]

	module.reducer = Rodux.createReducer(v25, {
		[p2 .. "_OWNED_GIVE"] = function(p3, p4)
			local payload = p4.payload
			local cosmeticId = payload.cosmeticId
			local amount = payload.amount
			local customValue = payload.customValue
			local shallowCopyTable = TableUtil.shallowCopyTable(p3)

			if customValue then
				shallowCopyTable[cosmeticId] = customValue

				return shallowCopyTable
			end

			if amount then
				local v43 = (shallowCopyTable[cosmeticId] or 0) + amount

				shallowCopyTable[cosmeticId] = if v43 > 0 then v43 else nil

				return shallowCopyTable
			end

			if not shallowCopyTable[cosmeticId] then
				shallowCopyTable[cosmeticId] = 1
			end

			return shallowCopyTable
		end,
		[p2 .. "_OWNED_REMOVE"] = function(p5, p6)
			local payload = p6.payload
			local cosmeticId = payload.cosmeticId
			local amount = payload.amount
			local shallowCopyTable2 = TableUtil.shallowCopyTable(p5)

			if amount then
				local v50 = (shallowCopyTable2[cosmeticId] or 0) - amount

				shallowCopyTable2[cosmeticId] = if v50 > 0 then v50 else nil

				return shallowCopyTable2
			end

			shallowCopyTable2[cosmeticId] = nil

			return shallowCopyTable2
		end
	})
	module.key = ownedThingsId

	return module
end
local function createEquippedCosmeticReducerInfo(p7)
	local equippedThingId = individualTypeConfigs[p7].equippedThingId
	local module4 = {}
	local v29 = DefaultData[equippedThingId]

	module4.reducer = Rodux.createReducer(v29, {
		[p7 .. "_EQUIPPED_CHANGE"] = function(_, p9)
			return p9.payload
		end
	})
	module4.key = equippedThingId

	return module4
end
local function createExtraCosmeticsDataReducerInfo(p10)
	local extraThingDataId = typeConfigs[p10].extraThingDataId

	if not extraThingDataId then
		return
	end

	local module3 = {}
	local v33 = DefaultData[extraThingDataId]
	local v34 = ThingUtil.createExtraThingDataActionHandlers(p10, CosmeticGroups.typeConfigs, CosmeticGroups.individualTypeConfigs)

	module3.reducer = Rodux.createReducer(v33, v34)
	module3.key = extraThingDataId

	return module3
end

local module2 = {}

local function onSingleReducerInfoCreated(reducer)
	if not reducer then
		return
	end

	module2[reducer.key] = reducer.reducer
end

for k, _ in pairs(typeConfigs) do
	local v16 = createOwnedCosmeticsReducerInfo(k)

	if v16 then
		module2[v16.key] = v16.reducer
	end

	local v17 = createExtraCosmeticsDataReducerInfo(k)

	if v17 then
		module2[v17.key] = v17.reducer
	end
end

for k, _ in pairs(individualTypeConfigs) do
	local v20 = createEquippedCosmeticReducerInfo(k)

	if v20 then
		module2[v20.key] = v20.reducer
	end
end

return module2
