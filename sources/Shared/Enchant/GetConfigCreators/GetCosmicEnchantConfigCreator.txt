local Cosmic = game.ReplicatedStorage.Shared.Assets.Cosmetics.Enchants.Cosmic

return function(config)
	local color = config.color
	local color2 = config.color2

	return function()
		return {
			displayName = config.displayName,
			icon = config.icon,
			apply = function(target)
				local EnchantsUtil = require(game.ReplicatedStorage.Shared.Source.Enchant.EnchantsUtil)

				local function onVisualCreated(visual)
					if visual:IsA("ParticleEmitter") then
						if visual.Name == "ParticleEmitter" then
							if color then
								visual.Color = ColorSequence.new(color)

								return
							end
						elseif color2 then
							visual.Color = ColorSequence.new(color2)
						end
					end
				end

				EnchantsUtil.runCallbackOnTaggedEnchantParts(target, function(part)
					EnchantsUtil.applyVisualsToPart(part, {
						scale = part.Size.Magnitude / 4
					}, Cosmic, onVisualCreated)
				end)
			end,
			vaulted = false,
			refundIfVaulted = false,
			rarity = "epic",
			inItemShop = true
		}
	end
end
