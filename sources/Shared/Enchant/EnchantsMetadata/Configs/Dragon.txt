local Dragon = game.ReplicatedStorage.Shared.Assets.Cosmetics.Enchants.Dragon

return function()
	return {
		displayName = "Dragon",
		icon = "rbxassetid://11868585634",
		apply = function(target)
			local EnchantsUtil = require(game.ReplicatedStorage.Shared.Source.Enchant.EnchantsUtil)

			EnchantsUtil.runCallbackOnTaggedEnchantParts(target, function(part)
				EnchantsUtil.applyVisualsToPart(part, {
					scale = part.Size.Magnitude / 8
				}, Dragon)
			end)
		end,
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true,
		caseIds = nil
	}
end
