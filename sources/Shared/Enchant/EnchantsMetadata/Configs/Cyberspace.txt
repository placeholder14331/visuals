local Cyberspace = game.ReplicatedStorage.Shared.Assets.Cosmetics.Enchants.Cyberspace

return function()
	return {
		displayName = "Cyberspace",
		icon = "rbxassetid://11897803782",
		apply = function(target)
			local EnchantsUtil = require(game.ReplicatedStorage.Shared.Source.Enchant.EnchantsUtil)
			local SkinUtil = require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil)

			EnchantsUtil.runCallbackOnTaggedEnchantParts(target, function(part)
				part.Material = Enum.Material.Neon
				part.Color = Color3.fromRGB(255, 111, 111)
				SkinUtil.handleTextureChangedOnSkinInst(part)
				EnchantsUtil.applyVisualsToPart(part, {
					scale = part.Size.Magnitude / 3
				}, Cyberspace)
			end)
		end,
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true,
		caseIds = nil
	}
end
