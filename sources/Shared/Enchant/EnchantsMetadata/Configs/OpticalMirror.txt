local OpticalMirror = game.ReplicatedStorage.Shared.Assets.Cosmetics.Enchants.OpticalMirror

return function()
	return {
		displayName = "Optical Mirror",
		icon = "rbxassetid://80123977299581",
		apply = function(target)
			local EnchantsUtil = require(game.ReplicatedStorage.Shared.Source.Enchant.EnchantsUtil)

			EnchantsUtil.runCallbackOnTaggedEnchantParts(target, function(part)
				EnchantsUtil.applyVisualsToPart(part, {
					scale = part.Size.Magnitude / 3.85
				}, OpticalMirror)
			end)
		end,
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true
	}
end
