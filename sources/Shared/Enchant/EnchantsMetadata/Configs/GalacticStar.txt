local GalacticStar = game.ReplicatedStorage.Shared.Assets.Cosmetics.Enchants.GalacticStar

return function()
	return {
		displayName = "Galactic Star",
		icon = "rbxassetid://9726015989",
		apply = function(target)
			local EnchantsUtil = require(game.ReplicatedStorage.Shared.Source.Enchant.EnchantsUtil)

			EnchantsUtil.runCallbackOnTaggedEnchantParts(target, function(part)
				EnchantsUtil.applyVisualsToPart(part, {
					scale = part.Size.Magnitude / 6
				}, GalacticStar)
			end)
		end,
		vaulted = false,
		refundIfVaulted = false,
		rarity = "rare",
		inItemShop = true
	}
end
