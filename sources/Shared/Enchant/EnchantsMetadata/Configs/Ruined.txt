local Ruined = game.ReplicatedStorage.Shared.Assets.Cosmetics.Enchants.Ruined

return function()
	return {
		displayName = "Ruin",
		icon = "rbxassetid://88484362503101",
		apply = function(target)
			local EnchantsUtil = require(game.ReplicatedStorage.Shared.Source.Enchant.EnchantsUtil)

			EnchantsUtil.runCallbackOnTaggedEnchantParts(target, function(part)
				EnchantsUtil.applyVisualsToPart(part, {
					scale = part.Size.Magnitude / 3
				}, Ruined)
			end)
		end,
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true,
		caseIds = nil
	}
end
