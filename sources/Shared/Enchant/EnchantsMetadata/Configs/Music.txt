local Music = game.ReplicatedStorage.Shared.Assets.Cosmetics.Enchants.Music

return function()
	return {
		displayName = "Music",
		icon = "rbxassetid://11897709458",
		apply = function(target)
			local EnchantsUtil = require(game.ReplicatedStorage.Shared.Source.Enchant.EnchantsUtil)

			EnchantsUtil.runCallbackOnTaggedEnchantParts(target, function(part)
				EnchantsUtil.applyVisualsToPart(part, {
					scale = part.Size.Magnitude / 4
				}, Music)
			end)
		end,
		vaulted = false,
		refundIfVaulted = false,
		rarity = "rare",
		inItemShop = true,
		caseIds = nil
	}
end
