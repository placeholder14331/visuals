local CollectionService = game:GetService("CollectionService")
local ScaleUtil = require(game.ReplicatedStorage.Shared.Source.Scale.ScaleUtil)
local CollectionServiceUtil = require(game.ReplicatedStorage.Shared.Source.CollectionService.CollectionServiceUtil)
local NumberUtil = require(game.ReplicatedStorage.Shared.Source.Number.NumberUtil)

return {
	applyVisualsToPart = function(part2, p2, part3, p4)
		local scale = (p2 or {}).scale
		local amount = NumberUtil.mapClamp(math.max(part2.Size.X, part2.Size.Z) / part2.Size.Y, 0, 1, 0.333, 1) * scale
		local v11 = if type(part3) ~= "table" then part3:GetChildren() else part3

		local function finalOnVisualCreated(p5)
			CollectionService:AddTag(p5, "EnchantVisuals_Client")

			if p4 then
				p4(p5)
			end
		end

		local module = {}

		local function getAttachmentClone(part)
			if not module[part] then
				local position = typeof(part3) == "Instance" and (not not part3:IsA("BasePart") and part2.Size / part3.Size) or Vector3.new(1, 1, 1)
				local Attachment = Instance.new("Attachment")

				Attachment.Position = part.Position * position
				Attachment.Orientation = part.Orientation
				CollectionService:AddTag(Attachment, "EnchantVisuals_Client")
				Attachment.Parent = part2
				module[part] = Attachment
			end

			return module[part]
		end
		local function applyVisual(visual)
			if visual:IsA("ParticleEmitter") then
				local clone = visual:Clone()

				if scale then
					ScaleUtil.scaleParticle(clone, scale, {
						scaleSize = false
					})
					clone.Size = ScaleUtil.scaleNumberSequence(clone.Size, amount)
				end

				if finalOnVisualCreated then
					CollectionService:AddTag(clone, "EnchantVisuals_Client")

					if p4 then
						p4(clone)
					end
				end

				clone.Parent = part2

				return
			end

			if visual:IsA("Trail") or visual:IsA("Beam") then
				local clone = visual:Clone()
				local p7Attachment0 = visual.Attachment0
				local v29 = p7Attachment0 and getAttachmentClone(p7Attachment0)
				local p7Attachment1 = visual.Attachment1
				local v31 = p7Attachment1 and getAttachmentClone(p7Attachment1)

				clone.Attachment0 = v29
				clone.Attachment1 = v31

				if finalOnVisualCreated then
					CollectionService:AddTag(clone, "EnchantVisuals_Client")

					if p4 then
						p4(clone)
					end
				end

				clone.Parent = part2

				return
			end

			if visual:IsA("Attachment") then
				for _, child in visual:GetChildren() do
					if child:IsA("ParticleEmitter") then
						local clone = child:Clone()

						if finalOnVisualCreated then
							CollectionService:AddTag(clone, "EnchantVisuals_Client")

							if p4 then
								p4(clone)
							end
						end

						clone.Parent = getAttachmentClone(visual)
					else
						applyVisual(child)
					end
				end
			end
		end

		for _, v in pairs(v11) do
			applyVisual(v)
		end
	end,
	runCallbackOnTaggedEnchantParts = function(container, p9)
		local descendants = container:GetDescendants()

		table.insert(descendants, container)
		CollectionServiceUtil.runCallbackOnTaggedInstances(descendants, {
			Enchant_Primary = p9
		}, function(part4)
			return part4:IsA("BasePart")
		end)
	end
}
