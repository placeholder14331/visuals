local ReplicatedStorage = game:GetService("ReplicatedStorage")
local CharacterHandler = require(ReplicatedStorage.Shared.Source.Character.CharacterHandler)

return function(_)
	return {
		signal = CharacterHandler.onPlayerDiedSignal,
		condition = function(p2, p3)
			return p3 == p2._character
		end,
		context = function()
			return {
				type = "death"
			}
		end
	}
end
