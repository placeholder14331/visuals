local SingleBatchedNumber = require(game.ReplicatedStorage.Shared.Source.Number.SingleBatchedNumber)

return {
	new = function()
		return {
			batchedNumbers = {},
			add = function(new4, new2, p3)
				if not new4.batchedNumbers[new2] then
					new4.batchedNumbers[new2] = SingleBatchedNumber.new()
				end

				new4.batchedNumbers[new2]:add(p3)
			end,
			subtract = function(new3, new5, p6)
				if not new3.batchedNumbers[new5] then
					new3.batchedNumbers[new5] = SingleBatchedNumber.new()
				end

				new3.batchedNumbers[new5]:subtract(p6)
			end,
			take = function(p7, p8)
				if not p7.batchedNumbers[p8] then
					return 0
				end

				return p7.batchedNumbers[p8]:take()
			end
		}
	end
}
