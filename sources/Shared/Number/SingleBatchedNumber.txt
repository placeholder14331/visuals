return {
	new = function()
		return {
			currAmnt = 0,
			add = function(currAmnt2, p2)
				currAmnt2.currAmnt = currAmnt2.currAmnt + p2
			end,
			subtract = function(currAmnt3, p4)
				currAmnt3.currAmnt = currAmnt3.currAmnt - p4
			end,
			take = function(currAmnt4)
				if math.abs(currAmnt4.currAmnt) < 1 then
					return 0
				end

				local amount = math.floor(currAmnt4.currAmnt)

				currAmnt4.currAmnt = currAmnt4.currAmnt - amount

				return amount
			end
		}
	end
}
