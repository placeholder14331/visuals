local Hexagon = game.ReplicatedStorage.Shared.Assets.Cosmetics.Skins.Hexagon

return function(config)
	local partColor = config.partColor
	local textureColor = config.textureColor
	local part2Color = config.part2Color
	local texture2Color = config.texture2Color

	return function()
		return {
			displayName = config.displayName,
			apply = function(target)
				local SkinUtil = require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil)

				local function onVisualCreated(p3)
					if p3:IsA("Texture") then
						SkinUtil.applyAnimation(p3)
						p3.Color3 = textureColor
					end
				end
				local function modifyPart(emitter2)
					emitter2.Color = partColor
				end
				local function onVisualCreated2(p5)
					if p5:IsA("Texture") then
						SkinUtil.applyAnimation(p5)
						p5.Color3 = texture2Color
					end
				end
				local function modifyPart2(emitter)
					emitter.Color = part2Color
				end

				SkinUtil.runCallbackOnTaggedSkinParts(target, function(emitter4)
					SkinUtil.applyVisualsToPart(emitter4, {}, Hexagon, onVisualCreated)
					emitter4.Color = partColor
				end, function(emitter3)
					SkinUtil.applyVisualsToPart(emitter3, {}, Hexagon, onVisualCreated2)
					emitter3.Color = part2Color
				end, function()
				end)
			end,
			vaulted = false,
			refundIfVaulted = false,
			rarity = "rare",
			inItemShop = true
		}
	end
end
