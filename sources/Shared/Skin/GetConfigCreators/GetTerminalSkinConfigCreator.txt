local Terminal = game.ReplicatedStorage.Shared.Assets.Cosmetics.Skins.Terminal

return function(config)
	local color = config.color

	return function()
		return {
			displayName = config.displayName,
			apply = function(target)
				local SkinUtil = require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil)

				local function onVisualCreated(p3)
					if p3:IsA("Texture") and color then
						p3.Color3 = color
					end
				end
				local function modifyPart(p4)
					p4.Material = Enum.Material.SmoothPlastic
					p4.Reflectance = 0
				end

				SkinUtil.runCallbackOnTaggedSkinParts(target, function(p5)
					SkinUtil.applyVisualsToPart(p5, {}, Terminal, onVisualCreated)
					p5.Material = Enum.Material.SmoothPlastic
					p5.Reflectance = 0
				end, function()
				end, function()
				end)
			end,
			vaulted = false,
			refundIfVaulted = false,
			rarity = "rare",
			inItemShop = true
		}
	end
end
