return function(config)
	local color = config.color
	local canObtain = config.canObtain == nil
	local animation = config.animation

	return function()
		return {
			displayName = config.displayName,
			apply = function(target)
				local SkinUtil = require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil)

				local function modifyPart(emitter3)
					if animation then
						SkinUtil.applyAnimation(emitter3, animation)
					end

					emitter3.Material = Enum.Material.Neon
					emitter3.Color = color
					emitter3.Reflectance = 0
				end

				SkinUtil.runCallbackOnTaggedSkinParts(target, function(emitter2)
					if animation then
						SkinUtil.applyAnimation(emitter2, animation)
					end

					emitter2.Material = Enum.Material.Neon
					emitter2.Color = color
					emitter2.Reflectance = 0
				end, function(emitter4)
					if animation then
						SkinUtil.applyAnimation(emitter4, animation)
					end

					emitter4.Material = Enum.Material.Neon
					emitter4.Color = color
					emitter4.Reflectance = 0
				end, function(emitter)
					if animation then
						SkinUtil.applyAnimation(emitter, animation)
					end

					emitter.Material = Enum.Material.Neon
					emitter.Color = color
					emitter.Reflectance = 0
				end)
			end,
			vaulted = false,
			refundIfVaulted = false,
			rarity = "legendary",
			inItemShop = canObtain,
			caseIds = not canObtain and {} or nil
		}
	end
end
