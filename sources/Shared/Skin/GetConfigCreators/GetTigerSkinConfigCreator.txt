local TigerStripes = game.ReplicatedStorage.Shared.Assets.Cosmetics.Skins.TigerStripes

return function(config)
	local partColor = config.partColor

	return function()
		return {
			displayName = config.displayName,
			apply = function(target)
				local SkinUtil = require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil)

				local function onVisualCreated(fromRGB)
					if fromRGB:IsA("Texture") then
						fromRGB.Color3 = Color3.fromRGB(0, 0, 0)
					end
				end
				local function modifyPart(emitter)
					emitter.Color = partColor
					emitter.Material = Enum.Material.Neon
				end

				SkinUtil.runCallbackOnTaggedSkinParts(target, function(emitter2)
					SkinUtil.applyVisualsToPart(emitter2, {}, TigerStripes, onVisualCreated)
					emitter2.Color = partColor
					emitter2.Material = Enum.Material.Neon
				end)
			end,
			vaulted = false,
			refundIfVaulted = false,
			rarity = "legendary",
			inItemShop = true
		}
	end
end
