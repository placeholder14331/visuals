local Floral = game.ReplicatedStorage.Shared.Assets.Cosmetics.Skins.Floral

return function(config)
	local textureColor = config.textureColor
	local partColor = config.partColor

	return function()
		return {
			displayName = config.displayName,
			apply = function(target)
				local SkinUtil = require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil)

				local function onVisualCreated(p3)
					if p3:IsA("Texture") and textureColor then
						p3.Color3 = textureColor
					end
				end
				local function modifyPart(emitter)
					emitter.Color = partColor
					emitter.Material = Enum.Material.SmoothPlastic
				end

				SkinUtil.runCallbackOnTaggedSkinParts(target, function(emitter2)
					SkinUtil.applyVisualsToPart(emitter2, {}, Floral, onVisualCreated)
					emitter2.Color = partColor
					emitter2.Material = Enum.Material.SmoothPlastic
				end, function()
				end, function()
				end)
			end,
			vaulted = false,
			refundIfVaulted = false,
			rarity = "rare",
			inItemShop = true
		}
	end
end
