local NeonCamo = game.ReplicatedStorage.Shared.Assets.Cosmetics.Skins.NeonCamo

return function(config)
	local color = config.color

	return function()
		return {
			displayName = config.displayName,
			apply = function(target)
				local SkinUtil = require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil)

				local function onVisualCreated(p3)
					if p3:IsA("Texture") and color then
						p3.Color3 = color
					end
				end
				local function modifyPart(emitter)
					emitter.Material = Enum.Material.Neon
					emitter.Color = color
				end

				SkinUtil.runCallbackOnTaggedSkinParts(target, function(emitter2)
					SkinUtil.applyVisualsToPart(emitter2, {}, NeonCamo, onVisualCreated)
					emitter2.Material = Enum.Material.Neon
					emitter2.Color = color
				end)
			end,
			vaulted = false,
			refundIfVaulted = false,
			rarity = "epic",
			inItemShop = true
		}
	end
end
