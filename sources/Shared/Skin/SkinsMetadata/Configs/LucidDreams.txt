local LucidDreams = game.ReplicatedStorage.Shared.Assets.Cosmetics.Skins.LucidDreams

return function()
	return {
		displayName = "Lucid Dreams",
		apply = function(target)
			local SkinUtil = require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil)

			local function onVisualCreated(p2)
				if p2:IsA("Texture") then
					SkinUtil.applyAnimation(p2)
				end
			end
			local function modifyPart(p3)
				p3.Material = Enum.Material.Neon
				p3.Reflectance = 0
			end

			SkinUtil.runCallbackOnTaggedSkinParts(target, function(p4)
				p4.Material = Enum.Material.Neon
				p4.Reflectance = 0
				SkinUtil.applyVisualsToPart(p4, {}, LucidDreams, onVisualCreated)
			end, function(p5)
				p5.Material = Enum.Material.Neon
				p5.Reflectance = 0
				SkinUtil.applyVisualsToPart(p5, {}, LucidDreams, onVisualCreated)
			end, function(p6)
				p6.Material = Enum.Material.Neon
				p6.Reflectance = 0
				SkinUtil.applyVisualsToPart(p6, {}, LucidDreams, onVisualCreated)
			end)
		end,
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true
	}
end
