return function()
	return {
		displayName = "Pumpkin Wrap",
		apply = function(target)
			local function modifyPart(p2)
				p2.Material = Enum.Material.Glass
				p2.Reflectance = 0.5
			end

			require(game.ReplicatedStorage.Shared.Source.Skin.SkinUtil).runCallbackOnTaggedSkinParts(target, function(emitter2)
				emitter2.Material = Enum.Material.Glass
				emitter2.Reflectance = 0.5
				emitter2.Color = Color3.fromHex("F46D0E")
			end, function(emitter)
				emitter.Material = Enum.Material.Glass
				emitter.Reflectance = 0.5
				emitter.Color = Color3.fromHex("F5D913")
			end, function(emitter3)
				emitter3.Material = Enum.Material.Glass
				emitter3.Reflectance = 0.5
				emitter3.Color = Color3.fromHex("2A5420")
			end)
		end,
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true
	}
end
