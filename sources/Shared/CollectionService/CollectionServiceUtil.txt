local CollectionService = game:GetService("CollectionService")

return {
	runCallbackOnTaggedInstances = function(p1, p2, p3)
		for _, v in pairs(p1) do
			if not p3 or p3(v) then
				for k, v2 in pairs(p2) do
					if CollectionService:HasTag(v, k) then
						v2(v)
					end
				end
			end
		end
	end
}
