local CollectionService = game:GetService("CollectionService")
local CharacterAuraConstants = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraConstants)
local VisualEffectsUtil = require(game.ReplicatedStorage.Shared.Source.VFX.VisualEffectsUtil)
local module = {
	onCharacterAuraInstanceCreated = function(p1)
		CollectionService:AddTag(p1, CharacterAuraConstants.CHARACTER_AURA_INSTANCE_TAG_NAME)
	end
}

function module.applyFromDummy(container, p3, p4)
	for _, child in pairs(container:GetChildren()) do
		local childName = p3:FindFirstChild(child.Name)

		if childName then
			for _, child2 in pairs(child:GetChildren()) do
				local clone = nil

				if child2:IsA("ParticleEmitter") or child2:IsA("Attachment") and #child2:GetChildren() > 0 then
					clone = child2:Clone()
					clone.Parent = childName
				elseif child2:IsA("Beam") or child2:IsA("Trail") then
					clone = VisualEffectsUtil.cloneOverAttachmentVisual(child2, childName)
				end

				if clone then
					module.onCharacterAuraInstanceCreated(clone)

					if p4 then
						if clone:IsA("Attachment") then
							for _, v16 in clone:QueryDescendants("ParticleEmitter, Beam, Trail") do
								p4(v16)
							end
						else
							p4(clone)
						end
					end
				end
			end
		end
	end
end

return module
