local Tendrils = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.Tendrils

return function()
	return {
		displayName = "Shadow Tendrils",
		icon = "rbxassetid://13515774924",
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true,
		apply = function(character)
			local CharacterAuraUtil = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil)
			local clone = Tendrils.Aura:Clone()

			CharacterAuraUtil.onCharacterAuraInstanceCreated(clone)
			clone.Parent = character

			local Weld = Instance.new("Weld")

			Weld.Part0 = character.HumanoidRootPart
			Weld.Part1 = clone
			Weld.C0 = CFrame.new(0, -2, 0) * CFrame.Angles(0, 0, -1.57079633)
			Weld.Parent = clone

			local clone2 = Tendrils.Part:Clone()

			CharacterAuraUtil.onCharacterAuraInstanceCreated(clone2)
			clone2.Parent = character

			local Weld2 = Instance.new("Weld")

			Weld2.Part0 = character.HumanoidRootPart
			Weld2.Part1 = clone2
			Weld2.Parent = clone2
		end
	}
end
