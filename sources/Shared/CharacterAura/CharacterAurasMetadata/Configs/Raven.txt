local Raven = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.Raven

return function()
	return {
		displayName = "Raven",
		icon = "rbxassetid://121684074323572",
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true,
		apply = function(character)
			require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil).applyFromDummy(Raven.Dummy, character)

			local clone = Raven.Ravens:Clone()

			clone.Parent = character
			clone.AnimationController:LoadAnimation(clone.Animation):Play()

			local Weld = Instance.new("Weld")

			Weld.Part0 = character.HumanoidRootPart
			Weld.Part1 = clone.PrimaryPart
			Weld.C0 = CFrame.new(0, -1.5, 0)
			Weld.Parent = character.HumanoidRootPart
		end
	}
end
