local DemonEyes = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.DemonEyes

return function()
	return {
		displayName = "Demon Eyes (Red)",
		icon = "rbxassetid://11572979661",
		vaulted = false,
		refundIfVaulted = false,
		rarity = "rare",
		inItemShop = true,
		apply = function(character)
			local CharacterAuraUtil = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil)
			local Torso = character:FindFirstChild("Torso")

			if Torso then
				local clone = DemonEyes.RedEyes:Clone()

				clone.Anchored = false
				CharacterAuraUtil.onCharacterAuraInstanceCreated(clone)

				local ManualWeld = Instance.new("ManualWeld")

				ManualWeld.Part0 = character.HumanoidRootPart
				ManualWeld.Part1 = clone
				ManualWeld.C0 = CFrame.new(0, 0, 0)
				ManualWeld.Parent = ManualWeld.Part1
				clone.Parent = Torso
			end
		end
	}
end
