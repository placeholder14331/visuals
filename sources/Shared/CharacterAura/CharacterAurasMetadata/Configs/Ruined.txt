local Ruined = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.Ruined

return function()
	return {
		displayName = "Ruin",
		icon = "rbxassetid://11572979332",
		vaulted = false,
		refundIfVaulted = false,
		rarity = "legendary",
		inItemShop = true,
		apply = function(character)
			local CharacterAuraUtil = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil)
			local clone = Ruined.Ruined:Clone()

			CharacterAuraUtil.onCharacterAuraInstanceCreated(clone)
			clone.Parent = character.HumanoidRootPart

			local Weld = Instance.new("Weld")

			Weld.Part0 = character.HumanoidRootPart
			Weld.Part1 = clone
			Weld.C0 = CFrame.new(0, -2.8, 0)
			Weld.Parent = clone
		end
	}
end
