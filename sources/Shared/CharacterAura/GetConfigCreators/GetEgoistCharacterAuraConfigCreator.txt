return function(config)
	local color = config.color
	local canObtain = config.canObtain == nil
	local rarity2 = config.rarity or "legendary"

	return function()
		return {
			displayName = config.displayName,
			icon = config.icon,
			vaulted = false,
			refundIfVaulted = false,
			rarity = rarity2,
			inItemShop = canObtain,
			caseIds = not canObtain and {} or nil,
			apply = function(character)
				local CharacterAuraUtil = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil)
				local Dummy = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.Egoist.Dummy

				CharacterAuraUtil.applyFromDummy(Dummy, character, function(visual)
					if (visual:IsA("ParticleEmitter") or (visual:IsA("Beam") or visual:IsA("Trail"))) and color then
						visual.Color = ColorSequence.new(color)
					end
				end)
			end
		}
	end
end
