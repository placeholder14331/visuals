local Fire = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.Fire

return function(config)
	local color = config.color
	local isDarkColor = config.isDarkColor
	local canObtain = config.canObtain == nil
	local rarity2 = config.rarity or "legendary"

	return function()
		return {
			displayName = config.displayName,
			icon = config.icon,
			vaulted = false,
			refundIfVaulted = false,
			rarity = rarity2,
			inItemShop = canObtain,
			caseIds = not canObtain and {} or nil,
			apply = function(character)
				local CharacterAuraUtil = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil)
				local clone = Fire.Part:Clone()

				CharacterAuraUtil.onCharacterAuraInstanceCreated(clone)
				clone.Parent = character

				local Weld = Instance.new("Weld")

				Weld.Part0 = character.Torso
				Weld.Part1 = clone
				Weld.C0 = CFrame.new(0, -0.5, 0)
				Weld.Parent = clone

				for _, child in pairs(clone:GetChildren()) do
					if child:IsA("ParticleEmitter") then
						if color then
							child.Color = ColorSequence.new(color)
						end

						child.LightEmission = not isDarkColor and 1 or 0
					end
				end
			end
		}
	end
end
