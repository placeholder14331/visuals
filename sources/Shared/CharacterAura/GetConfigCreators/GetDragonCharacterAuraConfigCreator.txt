local Dragon = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.Dragon
local Animation = Instance.new("Animation")

Animation.AnimationId = "rbxassetid://90492485620010"

return function(config)
	local prefabName = Dragon[config.prefabName]
	local canObtain = config.canObtain == nil
	local rarity2 = config.rarity or "legendary"

	return function()
		return {
			displayName = config.displayName,
			icon = config.icon,
			vaulted = false,
			refundIfVaulted = false,
			rarity = rarity2,
			inItemShop = canObtain,
			caseIds = not canObtain and {} or nil,
			apply = function(character)
				local CharacterAuraUtil = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil)
				local clone = prefabName.Part:Clone()

				CharacterAuraUtil.onCharacterAuraInstanceCreated(clone)

				local clone2 = prefabName.Wings2:Clone()

				CharacterAuraUtil.onCharacterAuraInstanceCreated(clone2)

				local Weld = Instance.new("Weld")

				Weld.Part0 = character.HumanoidRootPart
				Weld.Part1 = clone
				Weld.C0 = CFrame.new(0, -2.8, 0)
				Weld.Parent = Weld.Part0

				local Weld2 = Instance.new("Weld")

				Weld2.Part0 = clone2.RootPart
				Weld2.Part1 = character.Torso
				Weld2.C0 = CFrame.new(0, 5.75, -0.5)
				Weld2.Parent = Weld2.Part0
				clone.Parent = character.HumanoidRootPart
				clone2.Parent = character.Torso
				clone2.AnimationController.Animator:LoadAnimation(Animation):Play()
				CharacterAuraUtil.applyFromDummy(prefabName.Dummy, character)
			end
		}
	end
end
