local CosmicShield = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.CosmicShield

return function(config)
	local color = config.color
	local color2 = config.color2
	local isDarkColor = config.isDarkColor
	local canObtain = config.canObtain == nil
	local rarity2 = config.rarity or "epic"

	return function()
		return {
			displayName = config.displayName,
			icon = config.icon,
			vaulted = false,
			refundIfVaulted = false,
			rarity = rarity2,
			inItemShop = canObtain,
			caseIds = not canObtain and {} or nil,
			apply = function(character)
				require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil).applyFromDummy(CosmicShield, character, function(visual)
					if visual:IsA("ParticleEmitter") then
						if visual.Name == "ParticleEmitter" then
							if color then
								visual.Color = ColorSequence.new(color)
							end

							visual.LightEmission = not isDarkColor and 1 or 0

							return
						end

						if color2 then
							visual.Color = ColorSequence.new(color2)
						end

						visual.LightEmission = not isDarkColor and 1 or 0
					end
				end)
			end
		}
	end
end
