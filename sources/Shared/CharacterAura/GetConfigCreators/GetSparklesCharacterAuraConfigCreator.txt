local Sparkles = game.ReplicatedStorage.Shared.Assets.Cosmetics.CharacterAuras.Sparkles

return function(config)
	local color = config.color
	local isDarkColor = config.isDarkColor
	local canObtain = config.canObtain == nil
	local rarity2 = config.rarity or "rare"

	return function()
		return {
			displayName = config.displayName,
			icon = config.icon,
			vaulted = false,
			refundIfVaulted = false,
			rarity = rarity2,
			inItemShop = canObtain,
			caseIds = not canObtain and {} or nil,
			apply = function(character)
				require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraUtil).applyFromDummy(Sparkles.Dummy, character, function(visual)
					if visual:IsA("ParticleEmitter") then
						if color then
							visual.Color = ColorSequence.new(color)
						end

						visual.LightEmission = not isDarkColor and 1 or 0
					end
				end)
			end
		}
	end
end
