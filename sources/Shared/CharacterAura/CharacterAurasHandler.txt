local CollectionService = game:GetService("CollectionService")
local RunService = game:GetService("RunService"):IsServer()
local CharacterAurasMetadata = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAurasMetadata)
local CharacterAuraConstants = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraConstants)
local Network = require(game.ReplicatedStorage.Shared.Vendor.Network)

return {
	applyCharacterAura = function(p1, p2)
		if RunService then
			Network:FireAllClients("ApplyCharacterAura", p1, p2)

			return
		end

		local v8 = CharacterAurasMetadata[p2]

		if v8 and v8.apply then
			task.spawn(function()
				v8.apply(p1)
			end)
		end

		p1:SetAttribute("CharacterAuraId", p2)
	end,
	destroyCharacterAuras = function(container)
		for _, descendant in pairs(container:GetDescendants()) do
			if CollectionService:HasTag(descendant, CharacterAuraConstants.CHARACTER_AURA_INSTANCE_TAG_NAME) then
				descendant:Destroy()
			end
		end

		container:SetAttribute("CharacterAuraId", nil)
	end
}
