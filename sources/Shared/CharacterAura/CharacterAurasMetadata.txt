local CharacterAurasInOrder = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAurasInOrder)
local ThingUtil = require(game.ReplicatedStorage.Shared.Source.Thing.ThingUtil)
local CharacterAuraConstants = require(game.ReplicatedStorage.Shared.Source.CharacterAura.CharacterAuraConstants)
local CaseIds = require(game.ReplicatedStorage.Shared.Source.Case.CaseIds)
local Configs = script.Configs
local module = {}

ThingUtil.insertConfigs({
	handleCosmeticDefaults = true,
	cosmeticType = "characterAura",
	tblToInsert = module,
	configsToInsertInOrder = CharacterAurasInOrder,
	configsFolder = Configs,
	singleConfigCallback = function(config)
		if not config.aetherPrice then
			local n1 = 1e999

			if config.rarity == "legendary" then
				n1 = CharacterAuraConstants.LEGENDARY_PRICE
			elseif config.rarity == "epic" then
				n1 = CharacterAuraConstants.EPIC_PRICE
			elseif config.rarity == "rare" then
				n1 = CharacterAuraConstants.RARE_PRICE
			elseif config.rarity == "uncommon" then
				n1 = CharacterAuraConstants.UNCOMMON_PRICE
			end

			config.aetherPrice = n1
		end

		if not config.caseIds then
			config.caseIds = { CaseIds.characterAuras }
		end
	end
})

return module
