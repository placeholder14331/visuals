local Rodux = require(game.ReplicatedStorage.Shared.Vendor.Rodux)
local TableUtil = require(game.ReplicatedStorage.Shared.Vendor.Table.TableUtil)
local module = {}
local t2 = {
	isRagdolled = false
}

module.reducer = Rodux.createReducer(t2, {
	RAGDOLL_CLIENT_IS_RAGDOLLED_CHANGE = function(p1, p2)
		local shallowCopyTable = TableUtil.shallowCopyTable(p1)

		shallowCopyTable.isRagdolled = p2.payload

		return shallowCopyTable
	end
})
module.key = "ragdollClient"

return module
