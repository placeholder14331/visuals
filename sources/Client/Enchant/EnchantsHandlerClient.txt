local CollectionService = game:GetService("CollectionService")
local LocalPlayer = game.Players.LocalPlayer
local Network = require(game.ReplicatedStorage.Shared.Vendor.Network)
local EnchantsMetadata = require(game.ReplicatedStorage.Shared.Source.Enchant.EnchantsMetadata)
local RoduxStore = require(game.ReplicatedStorage.Shared.Source.Rodux.RoduxStore)
local Maid = require(game.ReplicatedStorage.Shared.Vendor.Maid)
local module = {
	removeEnchants = function(container)
		for _, descendant in pairs(container:GetDescendants()) do
			if CollectionService:HasTag(descendant, "EnchantVisuals_Client") then
				descendant:Destroy()
			end
		end

		container:SetAttribute("EnchantId", nil)
	end
}

local function getShouldDisableVfx(visual2)
	return (visual2:IsA("ParticleEmitter") or (visual2:IsA("Beam") or visual2:IsA("Trail"))) and (visual2.Enabled and not visual2:GetAttribute("alwaysKeep"))
end

function module.disableEnchants(container2)
	local maid3 = Maid.new()

	for _, descendant in container2:GetDescendants() do
		if (descendant:IsA("ParticleEmitter") or (descendant:IsA("Beam") or descendant:IsA("Trail"))) and (descendant.Enabled and not descendant:GetAttribute("alwaysKeep")) then
			descendant.Enabled = false
			maid3:GiveTask(function()
				descendant.Enabled = true
			end)
		end
	end

	return maid3
end
function module.apply(p4, p5, p6)
	local character = p6 or LocalPlayer.Character

	if not RoduxStore.store:getState().settingsClient.EnableOtherEnchants and character ~= LocalPlayer.Character then
		return
	end

	local v27 = EnchantsMetadata[p4]

	if v27 and v27.apply then
		v27.apply(p5, character)
	end

	p5:SetAttribute("EnchantId", p4)
end

local function removeItemsAttachedEnchants(p7, maid2)
	if not p7:FindFirstChild("ItemsAttached") then
		return
	end

	for _, child in pairs(p7.ItemsAttached:GetChildren()) do
		maid2:GiveTask(module.disableEnchants(child.Contents))
	end
end

local createStateToggler2 = Maid.createStateToggler()

local function onRangedCurrAimInfoChanged(aimInfo)
	if aimInfo and aimInfo.isAiming then
		createStateToggler2:create(function(maid)
			maid:GiveTask(module.disableEnchants(aimInfo.tool.Contents))

			if LocalPlayer.Character then
				removeItemsAttachedEnchants(LocalPlayer.Character, maid)
			end
		end)

		return
	end

	createStateToggler2:destroy()
end

local createStateToggler = Maid.createStateToggler()

local function onIsInFirstPersonChanged(p11)
	if p11 then
		createStateToggler:create(function(p12)
			if LocalPlayer.Character then
				removeItemsAttachedEnchants(LocalPlayer.Character, p12)
			end
		end)

		return
	end

	createStateToggler:destroy()
end
local function onEnchantVisualAdded(visual)
	if (not LocalPlayer.Backpack or not visual:IsDescendantOf(LocalPlayer.Backpack)) and ((not LocalPlayer.Character or not visual:IsDescendantOf(LocalPlayer.Character)) and visual:IsA("ParticleEmitter")) then
		visual.Rate = visual.Rate * 0.5
	end
end

function module._startModule()
	Network:BindEvents({
		ApplyEnchant = module.apply
	})

	local currAimInfo = RoduxStore.store:getState().rangedClient.currAimInfo

	if currAimInfo and currAimInfo.isAiming then
		createStateToggler2:create(function(maid4)
			maid4:GiveTask(module.disableEnchants(currAimInfo.tool.Contents))

			if LocalPlayer.Character then
				removeItemsAttachedEnchants(LocalPlayer.Character, maid4)
			end
		end)
	else
		createStateToggler2:destroy()
	end

	if RoduxStore.store:getState().firstPersonClient.isIn then
		createStateToggler:create(function(p15)
			if LocalPlayer.Character then
				removeItemsAttachedEnchants(LocalPlayer.Character, p15)
			end
		end)
	else
		createStateToggler:destroy()
	end

	for _, v in pairs(CollectionService:GetTagged("EnchantVisuals_Client")) do
		task.spawn(function()
			onEnchantVisualAdded(v)
		end)
	end

	RoduxStore.store:listenToKey("rangedClient", function(p16)
		return p16.rangedClient.currAimInfo
	end, onRangedCurrAimInfoChanged)
	RoduxStore.store:listenToKey("firstPersonClient", function(p17)
		return p17.firstPersonClient.isIn
	end, onIsInFirstPersonChanged)
	CollectionService:GetInstanceAddedSignal("EnchantVisuals_Client"):Connect(onEnchantVisualAdded)
end

return module
