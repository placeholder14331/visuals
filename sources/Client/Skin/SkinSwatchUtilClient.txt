local module = {}
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local SkinsMetadata = require(ReplicatedStorage.Shared.Source.Skin.SkinsMetadata)
local color3 = Color3.fromRGB(99, 95, 98)
local SmoothPlastic = Enum.Material.SmoothPlastic
local t2 = {
	primary = "Skin_Primary",
	secondary = "Skin_Secondary"
}
local t3 = {
	[Enum.Material.Metal] = true,
	[Enum.Material.DiamondPlate] = true,
	[Enum.Material.Foil] = true,
	[Enum.Material.Glass] = true,
	[Enum.Material.Ice] = true
}
local t4 = {
	[Enum.Material.Neon] = true,
	[Enum.Material.ForceField] = true
}
local t5 = {}

local function createProbe()
	local Model = Instance.new("Model")

	for v16, v17 in t2 do
		local Part = Instance.new("Part")

		Part.Name = v16
		Part.Size = Vector3.new(4, 1, 1)
		Part.Color = color3
		Part.Material = SmoothPlastic
		Part:AddTag(v17)
		Part.Parent = Model
	end

	return Model
end
local function readSurface(emitter)
	local t6 = {}

	for _, child in emitter:GetChildren() do
		if (child:IsA("Texture") or child:IsA("Decal")) and child.Face == Enum.NormalId.Front then
			table.insert(t6, {
				image = child.Texture,
				color = child.Color3,
				transparency = child.Transparency,
				tileStuds = if not child:IsA("Texture") then nil else child.StudsPerTileU
			})
		end
	end

	return {
		color = emitter.Color,
		material = emitter.Material,
		transparency = emitter.Transparency,
		layers = t6
	}
end
local function getIsUntouched(p2)
	return p2.color == color3 and (p2.material == SmoothPlastic and #p2.layers == 0)
end
local function getIsSameSurface(p3, p4)
	if p3.color ~= p4.color or (p3.material ~= p4.material or #p3.layers ~= #p4.layers) then
		return false
	end

	for v26, v27 in p3.layers do
		if v27.image ~= p4.layers[v26].image or v27.color ~= p4.layers[v26].color then
			return false
		end
	end

	return true
end
local function drawSurface(p5, p6)
	local color = p6.color

	if t4[p6.material] then
		color = color:Lerp(Color3.new(1, 1, 1), 0.25)
	end

	p5.BackgroundColor3 = color
	p5.BackgroundTransparency = math.min(p6.transparency, 0.6)

	for v31, v32 in p6.layers do
		local ImageLabel = Instance.new("ImageLabel")

		ImageLabel.Name = `Layer{ v31 }`
		ImageLabel.BackgroundTransparency = 1
		ImageLabel.Size = UDim2.fromScale(1, 1)
		ImageLabel.Image = v32.image
		ImageLabel.ImageColor3 = v32.color
		ImageLabel.ImageTransparency = v32.transparency
		ImageLabel.ZIndex = v31

		if v32.tileStuds then
			local amount = math.max(v32.tileStuds * 28, 8)

			ImageLabel.ScaleType = Enum.ScaleType.Tile
			ImageLabel.TileSize = UDim2.fromOffset(amount, amount)
		else
			ImageLabel.ScaleType = Enum.ScaleType.Crop
		end

		ImageLabel.Parent = p5
	end

	if t3[p6.material] then
		local Frame = Instance.new("Frame")

		Frame.Name = "Sheen"
		Frame.BackgroundColor3 = Color3.new(1, 1, 1)
		Frame.BorderSizePixel = 0
		Frame.Size = UDim2.fromScale(1, 1)
		Frame.ZIndex = #p6.layers + 1

		local UIGradient = Instance.new("UIGradient")

		UIGradient.Rotation = 35
		UIGradient.Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 1),
			NumberSequenceKeypoint.new(0.42, 1),
			NumberSequenceKeypoint.new(0.5, 0.72),
			NumberSequenceKeypoint.new(0.58, 1),
			NumberSequenceKeypoint.new(1, 1)
		})
		UIGradient.Parent = Frame
		Frame.Parent = p5

		return
	end

	if t4[p6.material] then
		local UIStroke = Instance.new("UIStroke")

		UIStroke.Name = "Glow"
		UIStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
		UIStroke.Color = color
		UIStroke.Thickness = 2
		UIStroke.Transparency = 0.2
		UIStroke.Parent = p5
	end
end

function module.getLook(p7)
	local v39 = t5[p7]

	if v39 then
		return v39
	end

	local instance = createProbe()
	local apply = SkinsMetadata[p7].apply

	if apply then
		apply(instance, nil)
	end

	local primary2 = readSurface(instance.primary)
	local secondary2 = readSurface(instance.secondary)

	instance:Destroy()

	local t7 = {
		primary = primary2,
		secondary = if not (secondary2.color == color3 and (secondary2.material == SmoothPlastic and #secondary2.layers == 0)) and not getIsSameSurface(primary2, secondary2) then secondary2 else nil
	}

	t5[p7] = t7

	return t7
end
function module.mount(p8, p9)
	local getLook = module.getLook(p9)
	local Frame = Instance.new("Frame")

	Frame.Name = "Swatch"
	Frame.BorderSizePixel = 0
	Frame.Size = UDim2.fromScale(1, 1)
	Frame.ClipsDescendants = true
	drawSurface(Frame, getLook.primary)

	local secondary = getLook.secondary

	if secondary then
		local Frame2 = Instance.new("Frame")

		Frame2.Name = "Secondary"
		Frame2.BorderSizePixel = 0
		Frame2.AnchorPoint = Vector2.new(0, 1)
		Frame2.Position = UDim2.fromScale(0, 1)
		Frame2.Size = UDim2.fromScale(1, 0.28)
		Frame2.ClipsDescendants = true
		Frame2.ZIndex = 20
		drawSurface(Frame2, secondary)
		Frame2.Parent = Frame
	end

	Frame.Parent = p8

	return function()
		Frame:Destroy()
	end
end

return module
