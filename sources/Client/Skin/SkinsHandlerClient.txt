local CollectionService = game:GetService("CollectionService")
local LocalPlayer = game.Players.LocalPlayer
local Network = require(game.ReplicatedStorage.Shared.Vendor.Network)
local SkinsMetadata = require(game.ReplicatedStorage.Shared.Source.Skin.SkinsMetadata)
local RoduxStore = require(game.ReplicatedStorage.Shared.Source.Rodux.RoduxStore)
local module = {
	removeSkins = function(container)
		for _, descendant in pairs(container:GetDescendants()) do
			if CollectionService:HasTag(descendant, "SkinVisual_Client") then
				descendant:Destroy()
			end
		end
	end,
	apply = function(target, p3, p4)
		local character = p4 or LocalPlayer.Character

		if not RoduxStore.store:getState().settingsClient.EnableOtherSkins and character ~= LocalPlayer.Character then
			return
		end

		local v14 = SkinsMetadata[target]

		if v14 and v14.apply then
			v14.apply(p3, character)
		end
	end
}

function module._startModule()
	Network:BindEvents({
		ApplySkin = module.apply
	})
end

return module
